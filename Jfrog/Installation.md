```bash
# 7. Go to binaries
cd $JFROG_HOME/artifactory/app/bin

# 8. Install as a system service
sudo ./installService.sh

# 9. Start Artifactory
sudo systemctl start artifactory

# 10. Check status
sudo systemctl status artifactory
```
